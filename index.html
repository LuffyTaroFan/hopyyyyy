    let currentQ = 0;  

    let score = 0;  

    let answered = false;  

  

    function renderQuiz() {  

      const q = quizData[currentQ];  

      const card = document.getElementById('quiz-card');  

        

      let optionsHTML = '';  

      q.options.forEach((opt, idx) => {  

        optionsHTML += `<button class="opt-btn" onclick="checkAnswer(${idx})">${opt}</button>`;  

      });  

  

      card.innerHTML = `  

        <div class="quiz-meta">  

          <span class="author-badge">👤 الفقرة: ${q.author}</span>  

          <span class="quiz-progress">السؤال ${currentQ + 1} من ${quizData.length}</span>  

        </div>  

        <div class="question-text">${q.q}</div>  

        <div class="options-grid" id="opts-container">${optionsHTML}</div>  

        <div class="explanation-box" id="exp-box"></div>  

        <div class="controls">  

          <button class="ctrl-btn" id="next-q-btn" onclick="nextQuestion()" style="display:none;">السؤال التالي ➔</button>  

        </div>  

      `;  

      answered = false;  

    }  

  

    function checkAnswer(selectedIdx) {  

      if(answered) return;  

      answered = true;  

  

      const q = quizData[currentQ];  

      const btns = document.querySelectorAll('#opts-container .opt-btn');  

      const expBox = document.getElementById('exp-box');  

      const nextBtn = document.getElementById('next-q-btn');  

  

      btns.forEach((btn, idx) => {  

        btn.disabled = true;  

        if(idx === q.correct) {  

          btn.classList.add('correct');  

        }  

        if(idx === selectedIdx && selectedIdx !== q.correct) {  

          btn.classList.add('wrong');  

        }  

      });  

  

      if(selectedIdx === q.correct) {  

        score++;  

      }  

  

      expBox.innerText = q.exp;  

      expBox.style.display = 'block';  

      nextBtn.style.display = 'block';  

    }  

  

    function nextQuestion() {  

      currentQ++;  

      if(currentQ < quizData.length) {  

        renderQuiz();  

      } else {  

        showFinalScore();  

      }  

    }  

  

    function showFinalScore() {  

      const card = document.getElementById('quiz-card');  

      card.innerHTML = `  

        <div class="score-view">  

          <h2>انتهى التحدي بنجاح! 🎉</h2>  

          <p style="color: var(--text-sub); margin-top: 10px;">إليك النتيجة النهائية لجميع الفقرات:</p>  

          <div class="score-badge">${score} / ${quizData.length}</div>  

          <button class="ctrl-btn" onclick="restartQuiz()">إعادة التحدي 🔄</button>  

        </div>  

      `;  

    }  

  

    function restartQuiz() {  

      currentQ = 0;  

      score = 0;  

      renderQuiz();  

    }  

  

    function toggleFullScreen() {  

      if (!document.fullscreenElement) {  

        document.documentElement.requestFullscreen();  

      } else {  

        if (document.exitFullscreen) {  

          document.exitFullscreen();  

        }  

      }  

    }  

  

    renderQuiz();
